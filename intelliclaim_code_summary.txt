from django.db import models
from django.contrib.auth.models import AbstractUser
from django.utils.translation import gettext_lazy as _


class UserRole(models.TextChoices):
    POLICYHOLDER = 'POLICYHOLDER', _('Policyholder')
    ADJUSTER = 'ADJUSTER', _('Adjuster')


class User(AbstractUser):
    """
    Custom user model with role-based access control.
    Supports Policyholders (filing claims) and Adjusters (evaluating & adjudicating claims).
    """
    role = models.CharField(
        max_length=20,
        choices=UserRole.choices,
        default=UserRole.POLICYHOLDER,
        help_text=_("Designates whether the user is a Policyholder or an Adjuster."),
    )
    phone_number = models.CharField(max_length=20, blank=True, null=True)
    policy_number = models.CharField(max_length=64, blank=True, null=True)

    @property
    def is_adjuster(self) -> bool:
        return self.role == UserRole.ADJUSTER or self.is_staff or self.is_superuser

    @property
    def is_policyholder(self) -> bool:
        return self.role == UserRole.POLICYHOLDER

    def __str__(self):
        return f"{self.username} ({self.get_role_display()})"


class ClaimStatus(models.TextChoices):
    PENDING = 'PENDING', _('Pending')
    UNDER_REVIEW = 'UNDER_REVIEW', _('Under Review')
    APPROVED = 'APPROVED', _('Approved')
    REJECTED = 'REJECTED', _('Rejected')


class CVRiskScore(models.TextChoices):
    LOW = 'Low', _('Low')
    HIGH = 'High', _('High')
    CRITICAL = 'Critical', _('Critical')


class NLPFraudRisk(models.TextChoices):
    LOW = 'Low', _('Low')
    MEDIUM = 'Medium', _('Medium')
    HIGH = 'High', _('High')


class Claim(models.Model):
    """
    Represents a vehicle insurance claim submitted for AI adjudication.
    Stores multi-modal AI inputs and assessments (OpenCV damage edge detection + Groq Llama 3 NLP).
    """
    user = models.ForeignKey(
        User,
        on_delete=models.CASCADE,
        related_name='claims',
        help_text=_("The policyholder submitting the claim.")
    )
    claim_id = models.CharField(
        max_length=32,
        unique=True,
        blank=True,
        help_text=_("Unique human-readable claim identifier (e.g. CLM-2025-001).")
    )
    description = models.TextField(
        help_text=_("Detailed incident narrative provided by the policyholder.")
    )
    image = models.ImageField(
        upload_to='claims/%Y/%m/%d/',
        help_text=_("Vehicle damage photograph.")
    )

    # OpenCV Assessment Fields
    cv_risk_score = models.CharField(
        max_length=20,
        choices=CVRiskScore.choices,
        default=CVRiskScore.LOW,
        help_text=_("Computer vision calculated physical damage risk severity.")
    )
    edge_density = models.FloatField(
        default=0.0,
        help_text=_("Calculated Canny edge density ratio from OpenCV.")
    )
    estimated_payout = models.DecimalField(
        max_digits=12,
        decimal_places=2,
        default=0.00,
        help_text=_("Estimated repair payout based on damage severity (in INR â‚¹).")
    )

    # Groq Llama 3 NLP Fraud Assessment Fields
    nlp_fraud_risk = models.CharField(
        max_length=20,
        choices=NLPFraudRisk.choices,
        default=NLPFraudRisk.LOW,
        help_text=_("LLM detected fraud probability level.")
    )
    nlp_reasoning = models.TextField(
        blank=True,
        help_text=_("Forensic justification generated by Llama 3 for the fraud assessment.")
    )

    # Adjudication Status & Lifecycle
    status = models.CharField(
        max_length=30,
        choices=ClaimStatus.choices,
        default=ClaimStatus.PENDING,
        help_text=_("Workflow status of the claim.")
    )
    adjuster_notes = models.TextField(
        blank=True,
        help_text=_("Optional remarks logged by the claims adjuster.")
    )

    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        ordering = ['-created_at']
        verbose_name = _('Vehicle Claim')
        verbose_name_plural = _('Vehicle Claims')

    def save(self, *args, **kwargs):
        if not self.claim_id:
            import uuid
            self.claim_id = f"CLM-{uuid.uuid4().hex[:8].upper()}"
        super().save(*args, **kwargs)

    def __str__(self):
        return f"{self.claim_id} - {self.user.username} - {self.status}"

def process_image_opencv(image_bytes: bytes) -> dict:
    """
    OpenCV Damage Severity Assessment Pipeline.
    
    1. Decodes uploaded image stream directly from memory.
    2. Converts to Grayscale.
    3. Applies Gaussian Blur with a (7, 7) kernel.
    4. Computes Canny Edge Detection (thresholds: 50, 150).
    5. Calculates edge density ratio = (non-zero edge pixels) / (total pixels).
    6. Adjudicates CV risk score and estimated payout in INR (â‚¹).
    """
    try:
        # Convert byte stream to numpy uint8 array
        np_arr = np.frombuffer(image_bytes, np.uint8)
        img = cv2.imdecode(np_arr, cv2.IMREAD_COLOR)

        if img is None:
            raise ValueError("Failed to decode image from memory buffer.")

        # 1. Convert to grayscale
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

        # 2. Gaussian Blur (7x7 kernel, sigma=0)
        blurred = cv2.GaussianBlur(gray, (7, 7), 0)

        # 3. Canny Edge Detection (50, 150)
        edges = cv2.Canny(blurred, 50, 150)

        # 4. Calculate edge density
        total_pixels = edges.size
        edge_pixels = np.count_nonzero(edges)
        edge_density = float(edge_pixels / total_pixels) if total_pixels > 0 else 0.0

        # Round edge density for clean presentation
        edge_density = round(edge_density, 5)

        # 5. Adjudicate CV Risk & Estimated Repair Payout
        if edge_density > 0.08:
            cv_risk_score = 'Critical'
            estimated_payout = 85000.00
        elif edge_density > 0.03:
            cv_risk_score = 'High'
            estimated_payout = 45000.00
        else:
            cv_risk_score = 'Low'
            estimated_payout = 12500.00

        return {
            'cv_risk_score': cv_risk_score,
            'estimated_payout': estimated_payout,
            'edge_density': edge_density,
            'raw_edge_count': int(edge_pixels),
            'total_pixels': int(total_pixels),
        }

    except Exception as e:
        logger.error(f"Error in OpenCV pipeline: {e}", exc_info=True)
        # Safe fallback in case of corrupted image
        return {
            'cv_risk_score': 'Low',
            'estimated_payout': 12500.00,
            'edge_density': 0.015,
            'error': str(e)
        }


def analyze_description_groq(description: str, cv_risk: str = "Unknown", estimated_payout: float = 0.0) -> dict:
    """
    Groq Llama 3 NLP Fraud Detection Pipeline.
    
    Zero-shot prompt to detect physically impossible, contradictory, staged,
    or inflated vehicle claim fraud patterns.
    Enforces JSON output: {"fraud_risk": "Low" | "Medium" | "High", "reasoning": "..."}
    """
    groq_api_key = getattr(settings, 'GROQ_API_KEY', os.getenv('GROQ_API_KEY', ''))
    groq_model = getattr(settings, 'GROQ_MODEL', os.getenv('GROQ_MODEL', 'llama3-8b-8192'))

    # If Groq API key is present, invoke Groq Cloud API
    if groq_api_key and groq_api_key != 'gsk_your_groq_api_key_here':
        try:
            from groq import Groq
            client = Groq(api_key=groq_api_key)

            system_prompt = (
                "You are an expert Automated Vehicle Insurance Fraud Adjudicator for IntelliClaim. "
                "Analyze the vehicle collision/incident narrative and cross-reference with Computer Vision damage severity. "
                "Detect physically impossible physics, contradictory timeline narratives, staged accident hallmarks, "
                "vague descriptions of major crashes, or inflated payout demands.\n\n"
                "Return strictly valid JSON with no markdown wrapping or preamble, matching this exact schema:\n"
                "{\n"
                '  "fraud_risk": "Low" | "Medium" | "High",\n'
                '  "reasoning": "Detailed technical and forensic explanation of the verdict"\n'
                "}"
            )

            user_prompt = (
                f"Incident Narrative: \"{description}\"\n"
                f"CV Assessed Damage Severity: {cv_risk}\n"
                f"Estimated Repair Payout: INR â‚¹{estimated_payout:,.2f}\n\n"
                "Assess whether this claim contains fraud anomalies, physical contradictions, or staged accident patterns."
            )

            chat_completion = client.chat.completions.create(
                messages=[
                    {"role": "system", "content": system_prompt},
                    {"role": "user", "content": user_prompt}
                ],
                model=groq_model,
                temperature=0.1,
                response_format={"type": "json_object"},
            )

            raw_response = chat_completion.choices[0].message.content
            parsed = json.loads(raw_response)

            fraud_risk = parsed.get("fraud_risk", "Low")
            # Normalize casing
            if fraud_risk.capitalize() in ["Low", "Medium", "High"]:
                fraud_risk = fraud_risk.capitalize()
            else:
                fraud_risk = "Low"

            reasoning = parsed.get("reasoning", "No suspicious anomalies or physical impossibilities detected in the incident report.")

            return {
                "fraud_risk": fraud_risk,
                "reasoning": reasoning
            }

        except Exception as exc:
            logger.warning(f"Groq API call encountered an error: {exc}. Falling back to internal heuristic engine.")

    # Intelligent fallback analyzer (guarantees uptime even if no internet/API key configured)
    return _heuristic_fraud_analysis(description, cv_risk)


def _heuristic_fraud_analysis(description: str, cv_risk: str) -> dict:
    """
    Robust rule-based heuristic fallback if Groq API is unavailable or unconfigured.
    """
    desc_lower = description.lower()
    
    suspicious_keywords = [
        "totaled", "brand new", "cash settlement", "staged", "phantom", "ghost vehicle",
        "hit and run", "empty parking lot", "no witnesses", "unattended", "midnight",
        "abandoned", "keys left inside", "mysterious fire", "submerged in pond"
    ]
    
    inconsistent_severity = (
        ("scratched" in desc_lower or "minor dent" in desc_lower) and cv_risk == "Critical"
    ) or (
        ("catastrophic rollover" in desc_lower or "engine block crushed" in desc_lower) and cv_risk == "Low"
    )

    matches = [word for word in suspicious_keywords if word in desc_lower]

    if inconsistent_severity or len(matches) >= 3:
        return {
            "fraud_risk": "High",
            "reasoning": f"Flagged high anomaly: Critical narrative/damage mismatch detected. "
                         f"Narrative indicators matched: {', '.join(matches) if matches else 'physical damage inconsistency'}. "
                         "Requires manual investigation by senior special claims unit."
        }
    elif len(matches) in [1, 2]:
        return {
            "fraud_risk": "Medium",
            "reasoning": f"Moderate fraud risk: Unwitnessed scenario noted with keywords ({', '.join(matches)}). "
                         "Cross-examination with vehicular telemetry and road CCTV recommended."
        }
    else:
        return {
            "fraud_risk": "Low",
            "reasoning": "Consistent vehicular physics and plausible incident narrative with no temporal or structural discrepancies detected."
        }

class ClaimAssessView(APIView):
    """
    POST /api/claims/assess/
    Multi-modal AI Adjudication Endpoint.
    
    Accepts multipart/form-data containing:
      - 'image': Vehicle damage photograph
      - 'description': Textual narrative of the accident
      - 'policy_number' (optional)
      
    Pipeline:
      1. OpenCV: In-memory image processing -> Grayscale -> Gaussian Blur (7x7) ->
                 Canny (50, 150) -> Edge density calculation -> Damage severity & repair payout.
      2. Groq LLM: Zero-shot prompt with Llama 3 8B -> Fraud anomaly detection & reasoning.
      3. Persistence: Atomically saves claim record in PostgreSQL database.
      4. Response: Returns complete multi-modal assessment payload.
    """
    parser_classes = [MultiPartParser, FormParser]
    permission_classes = [permissions.AllowAny]  # Allows authenticated users or demo guest submissions

    @transaction.atomic
    def post(self, request, *args, **kwargs):
        serializer = ClaimAssessInputSerializer(data=request.data)
        if not serializer.is_valid():
            return Response(
                {"error": "Validation failed", "details": serializer.errors},
                status=status.HTTP_400_BAD_REQUEST
            )

        validated_data = serializer.validated_data
        image_file = validated_data['image']
        description = validated_data['description']

        # Determine user context (Authenticated or Demo Policyholder fallback)
        if request.user and request.user.is_authenticated:
            claim_user = request.user
        else:
            # Create or get demo policyholder user for convenience in tests/evaluations
            claim_user, _ = User.objects.get_or_create(
                username='demo_policyholder',
                defaults={
                    'email': 'policyholder@intelliclaim.ai',
                    'role': UserRole.POLICYHOLDER,
                    'first_name': 'Demo',
                    'last_name': 'Policyholder',
                    'policy_number': 'POL-IND-884920'
                }
            )

        try:
            # 1. OpenCV In-Memory Pipeline
            image_file.seek(0)
            image_bytes = image_file.read()
            cv_results = process_image_opencv(image_bytes)

            cv_risk_score = cv_results.get('cv_risk_score', 'Low')
            estimated_payout = cv_results.get('estimated_payout', 12500.00)
            edge_density = cv_results.get('edge_density', 0.0)

            # 2. Groq Llama 3 NLP Fraud Detection Pipeline
            nlp_results = analyze_description_groq(
                description=description,
                cv_risk=cv_risk_score,
                estimated_payout=estimated_payout
            )

            nlp_fraud_risk = nlp_results.get('fraud_risk', 'Low')
            nlp_reasoning = nlp_results.get('reasoning', '')

            # 3. Save Claim into PostgreSQL database
            image_file.seek(0)
            claim = Claim.objects.create(
                user=claim_user,
                description=description,
                image=image_file,
                cv_risk_score=cv_risk_score,
                edge_density=edge_density,
                estimated_payout=estimated_payout,
                nlp_fraud_risk=nlp_fraud_risk,
                nlp_reasoning=nlp_reasoning,
                status=ClaimStatus.PENDING,
            )

            output_serializer = ClaimSerializer(claim, context={'request': request})

            return Response({
                "message": "Claim successfully assessed and registered into adjudication pipeline.",
                "claim": output_serializer.data,
                "ai_diagnostics": {
                    "opencv_metrics": {
                        "cv_risk_score": cv_risk_score,
                        "edge_density": edge_density,
                        "estimated_payout": estimated_payout,
                        "raw_edge_count": cv_results.get('raw_edge_count'),
                        "total_pixels": cv_results.get('total_pixels'),
                    },
                    "nlp_metrics": {
                        "fraud_risk": nlp_fraud_risk,
                        "reasoning": nlp_reasoning,
                        "model": getattr(settings, 'GROQ_MODEL', 'llama3-8b-8192')
                    }
                }
            }, status=status.HTTP_201_CREATED)

        except Exception as exc:
            logger.error(f"Adjudication pipeline error: {exc}", exc_info=True)
            return Response(
                {"error": "Failed to complete AI adjudication.", "details": str(exc)},
                status=status.HTTP_500_INTERNAL_SERVER_ERROR
            )


class ClaimListView(generics.ListAPIView):
    """
    GET /api/claims/
    Returns list of claims.
    - Adjusters and Admins see all claims.
    - Policyholders see only their own claims.
    """
    serializer_class = ClaimSerializer
    permission_classes = [permissions.IsAuthenticated]

    def get_queryset(self):
        user = self.request.user
        if user.is_adjuster:
            return Claim.objects.all().select_related('user')
        return Claim.objects.filter(user=user).select_related('user')

    def get_queryset(self):
        user = self.request.user
        if user.is_adjuster:

class ClaimStatusUpdateView(APIView):
    """
    PATCH /api/claims/<id>/status/
    Allows an Adjuster to adjudicate a claim (Approve, Reject, Under Review)
    and append optional notes.
    """
    permission_classes = [permissions.IsAuthenticated]

    def patch(self, request, id):
        if not request.user.is_adjuster:
            return Response(
                {"error": "Permission denied. Only claims adjusters can update claim status."},
                status=status.HTTP_403_FORBIDDEN
            )

        try:
            claim = Claim.objects.get(id=id)
        except Claim.DoesNotExist:
            return Response({"error": "Claim not found."}, status=status.HTTP_404_NOT_FOUND)

        serializer = ClaimStatusUpdateSerializer(claim, data=request.data, partial=True)
        if serializer.is_valid():
            serializer.save()
            return Response({
                "message": f"Claim status updated to '{claim.status}'.",
                "claim": ClaimSerializer(claim, context={'request': request}).data
            })
        return Response(serializer.errors, status=status.HTTP_400_BAD_REQUEST)

export interface AuthTokens {
  access: string;
  refresh: string;
  user: User;
}

import { AuthTokens, Claim, AssessmentResponse, ClaimStatus, User } from '../types';

const API_BASE_URL = import.meta.env.VITE_API_URL || 'http://localhost:8000/api';

class ApiService {
  private getAuthHeader(): Record<string, string> {
    const token = localStorage.getItem('intelliclaim_access_token');
    return token ? { Authorization: `Bearer ${token}` } : {};
  }

  // --- Auth APIs ---
  async login(username: string, password: string): Promise<AuthTokens> {
    const response = await fetch(`${API_BASE_URL}/auth/login/`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ username, password }),
    });

    if (!response.ok) {
      const err = await response.json().catch(() => ({}));
      throw new Error(err.detail || err.error || 'Authentication failed. Please verify credentials.');
    }

    const data: AuthTokens = await response.json();
    localStorage.setItem('intelliclaim_access_token', data.access);
    localStorage.setItem('intelliclaim_refresh_token', data.refresh);
    localStorage.setItem('intelliclaim_user', JSON.stringify(data.user));
    return data;
  }

  async getMe(): Promise<User> {
    const response = await fetch(`${API_BASE_URL}/auth/me/`, {
      headers: {
        'Content-Type': 'application/json',
        ...this.getAuthHeader(),
      },
    });

    if (!response.ok) {
      throw new Error('Failed to retrieve current user session');
    }
    return response.json();
  }

  logout(): void {
    localStorage.removeItem('intelliclaim_access_token');
    localStorage.removeItem('intelliclaim_refresh_token');
    localStorage.removeItem('intelliclaim_user');
  }

  // --- Claim Adjudication APIs ---
  async assessClaim(formData: FormData): Promise<AssessmentResponse> {
    const headers = this.getAuthHeader();
    // Do NOT set Content-Type header when posting FormData; browser must set boundary automatically
    const response = await fetch(`${API_BASE_URL}/claims/assess/`, {
      method: 'POST',
      headers,
      body: formData,
    });

    if (!response.ok) {
      const err = await response.json().catch(() => ({}));
      throw new Error(err.details?.image?.[0] || err.details?.description?.[0] || err.error || 'Claim adjudication processing failed.');
    }

    return response.json();
  }

  async getClaims(): Promise<Claim[]> {
    const response = await fetch(`${API_BASE_URL}/claims/`, {
      headers: {
        'Content-Type': 'application/json',
        ...this.getAuthHeader(),
      },
    });

    if (!response.ok) {
      throw new Error('Failed to fetch claims list');
    }
    return response.json();
  }

  async getClaimById(id: number): Promise<Claim> {
    const response = await fetch(`${API_BASE_URL}/claims/${id}/`, {
      headers: {
        'Content-Type': 'application/json',
        ...this.getAuthHeader(),
      },
    });

    if (!response.ok) {
      throw new Error(`Failed to fetch claim #${id}`);
    }
    return response.json();
  }

  async updateClaimStatus(id: number, status: ClaimStatus, notes?: string): Promise<{ message: string; claim: Claim }> {
    const response = await fetch(`${API_BASE_URL}/claims/${id}/status/`, {
      method: 'PATCH',
      headers: {
        'Content-Type': 'application/json',
        ...this.getAuthHeader(),
      },
      body: JSON.stringify({ status, adjuster_notes: notes || '' }),
    });

    if (!response.ok) {
      const err = await response.json().catch(() => ({}));
      throw new Error(err.error || 'Failed to update claim adjudication status');
    }
    return response.json();
  }

  async checkHealth(): Promise<{ status: string; database_connected: boolean; groq_configured: boolean }> {
    const response = await fetch(`${API_BASE_URL}/health/`);
    return response.json();
  }
}

export const api = new ApiService();
